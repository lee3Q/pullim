## 핵심 모듈 명세

### 1. providers/index.ts — 멀티 모델 프로바이더

```typescript
import { ModelProvider } from '@/lib/types-ultimate';

interface LLMResponse {
  text: string;
  usage: { input_tokens: number; output_tokens: number };
  latency_ms: number;
  model: string;
}

interface LLMProvider {
  chat(params: {
    system: string;
    messages: { role: 'user' | 'assistant'; content: string }[];
    maxTokens: number;
    temperature?: number;
  }): Promise<LLMResponse>;

  stream(params: {
    system: string;
    messages: { role: 'user' | 'assistant'; content: string }[];
    maxTokens: number;
  }): AsyncIterable<string>;
}

function getProvider(name: ModelProvider): LLMProvider;

// 사용 가능한 프로바이더 확인
function getAvailableProviders(): ModelProvider[];

// 병렬 호출
async function parallelChat(
  providers: { provider: ModelProvider; params: ChatParams }[]
): Promise<{ provider: ModelProvider; result: LLMResponse | Error }[]>;
```

## API 라우트 명세

### POST /api/ultimate/route

```typescript
// 입력
{ input: string }

// 출력
{
  sessionId: string;
  concernType: ConcernType;
  scores: RoutingScores;
  recommendedCrystals: CrystalName[];
  crisis?: { level: 'RED' | 'YELLOW'; message: string; hotlines: string[] };
}
```

### POST /api/ultimate/research

```typescript
// 입력
{ sessionId: string; concern: string; concernType: ConcernType }

// 출력 (SSE stream)
event: status
data: { stage: "querying" | "searching" | "structuring" | "verifying" | "done" }

event: result
data: ResearchOutput
```

### POST /api/ultimate/analyze

```typescript
// 입력
{
  sessionId: string;
  crystals: [CrystalName, CrystalName, CrystalName];
  concern: string;
  researchSummary: string;
  selectedDataCard?: number;
}

// 출력 (SSE stream — 분석 결과를 하나씩 스트리밍)
event: crystal
data: { crystal: CrystalName, model: ModelProvider, analysis: CrystalAnalysis }

event: disagreement
data: Disagreement[]

event: done
data: { totalCost: number }
```

### POST /api/ultimate/listen

```typescript
// V2와 동일한 SSE 스트리밍, 캐릭터 톤만 다름
// 입력
{ sessionId: string; message: string; character: CharacterName }

// 출력: SSE text stream
```

## Zustand 스토어 (ultimate-session-store.ts)

```typescript
interface UltimateSessionStore {
  // 세션
  sessions: Record<string, Session>;
  messages: Record<string, UltimateMessage[]>;

  // 액션
  createSession(theme: ThemeName, character: CharacterName): string;
  setRouting(id: string, ...): void;
  setResearch(id: string, data: ResearchOutput): void;
  setSelectedCrystals(id: string, crystals: [CrystalName, CrystalName, CrystalName]): void;
  setCrystalAnalyses(id: string, analyses: CrystalAnalysis[]): void;
  setDisagreements(id: string, points: Disagreement[]): void;
  setDebateRounds(id: string, rounds: DebateRound[]): void;
  setConclusion(id: string, data: ConclusionData): void;
  setActionCommitment(id: string, action: ActionCommitment): void;
  advanceStage(id: string): void;
  addMessage(id: string, msg: UltimateMessage): void;
  completeSession(id: string): void;

  // 조회
  getSessionCount(): number;
}
```

persist: localStorage (V2와 동일 패턴).
